import React, { useState } from "react";
import { StyleSheet, Text, View, FlatList, Image, TouchableOpacity, TextInput, ScrollView, Alert } from "react-native";

const SAMPLE_RESTAURANTS = [
  {
    id: "r1",
    name: "Masala Bite",
    cuisine: "Indian",
    eta: "25-35 min",
    rating: 4.5,
    priceLevel: "₹",
    image: "https://images.unsplash.com/photo-1604908177522-6f87c2d7f4f6",
    menu: [
      { id: "m1", name: "Butter Chicken", price: 320 },
      { id: "m2", name: "Paneer Tikka", price: 240 },
      { id: "m3", name: "Garlic Naan", price: 40 },
    ],
  },
  {
    id: "r2",
    name: "Sizzle Grill",
    cuisine: "BBQ",
    eta: "30-45 min",
    rating: 4.2,
    priceLevel: "₹₹",
    image: "https://images.unsplash.com/photo-1555992336-03a23c4d7f6b",
    menu: [
      { id: "m4", name: "Grilled Chicken", price: 380 },
      { id: "m5", name: "Pepper Steak", price: 420 },
      { id: "m6", name: "Fries", price: 120 },
    ],
  },
];

export default function App() {
  const [query, setQuery] = useState("");
  const [selected, setSelected] = useState(null);
  const [cartItems, setCartItems] = useState([]);

  const filtered = SAMPLE_RESTAURANTS.filter((r) => {
    const q = query.toLowerCase();
    return r.name.toLowerCase().includes(q) || r.cuisine.toLowerCase().includes(q);
  });

  function handleAdd(item) {
    const found = cartItems.find((c) => c.id === item.id);
    if (found) {
      setCartItems(cartItems.map((c) => c.id === item.id ? { ...c, qty: c.qty + 1 } : c));
    } else {
      setCartItems([...cartItems, { ...item, qty: 1 }]);
    }
  }

  function handleCheckout() {
    Alert.alert("Order placed!", "This is a demo checkout.");
    setCartItems([]);
  }

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <Text style={styles.logo}>🍔 FoodDrop</Text>
        <Text style={styles.cart}>🛒 {cartItems.reduce((s, c) => s + c.qty, 0)}</Text>
      </View>

      {/* Search */}
      <TextInput
        style={styles.search}
        placeholder="Search restaurants or dishes..."
        value={query}
        onChangeText={setQuery}
      />

      <ScrollView style={{ flex: 1 }}>
        {!selected ? (
          <>
            <Text style={styles.title}>Popular Restaurants</Text>
            <FlatList
              data={filtered}
              keyExtractor={(item) => item.id}
              renderItem={({ item }) => (
                <TouchableOpacity style={styles.card} onPress={() => setSelected(item)}>
                  <Image source={{ uri: item.image }} style={styles.image} />
                  <View style={{ flex: 1, marginLeft: 10 }}>
                    <Text style={styles.name}>{item.name}</Text>
                    <Text style={styles.sub}>{item.cuisine} • {item.eta}</Text>
                    <Text style={styles.sub}>⭐ {item.rating} • {item.priceLevel}</Text>
                  </View>
                </TouchableOpacity>
              )}
            />
          </>
        ) : (
          <View style={{ padding: 10 }}>
            <TouchableOpacity onPress={() => setSelected(null)}>
              <Text style={{ color: "blue", marginBottom: 5 }}>← Back</Text>
            </TouchableOpacity>
            <Text style={styles.title}>{selected.name}</Text>
            {selected.menu.map((m) => (
              <View key={m.id} style={styles.menuItem}>
                <Text>{m.name} - ₹{m.price}</Text>
                <TouchableOpacity style={styles.addBtn} onPress={() => handleAdd(m)}>
                  <Text style={{ color: "white" }}>Add</Text>
                </TouchableOpacity>
              </View>
            ))}
          </View>
        )}
      </ScrollView>

      {/* Cart */}
      {cartItems.length > 0 && (
        <View style={styles.cartBox}>
          <Text>Items: {cartItems.length}</Text>
          <TouchableOpacity style={styles.checkoutBtn} onPress={handleCheckout}>
            <Text style={{ color: "white" }}>Checkout</Text>
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: "#fff", paddingTop: 40 },
  header: { flexDirection: "row", justifyContent: "space-between", padding: 15 },
  logo: { fontSize: 20, fontWeight: "bold" },
  cart: { fontSize: 18 },
  search: { margin: 10, padding: 10, borderWidth: 1, borderRadius: 8 },
  title: { fontSize: 18, fontWeight: "bold", margin: 10 },
  card: { flexDirection: "row", padding: 10, margin: 10, backgroundColor: "#f9f9f9", borderRadius: 8 },
  image: { width: 80, height: 80, borderRadius: 8 },
  name: { fontSize: 16, fontWeight: "bold" },
  sub: { fontSize: 12, color: "#666" },
  menuItem: { flexDirection: "row", justifyContent: "space-between", paddingVertical: 10, borderBottomWidth: 1, borderColor: "#eee" },
  addBtn: { backgroundColor: "green", paddingHorizontal: 10, paddingVertical: 5, borderRadius: 5 },
  cartBox: { flexDirection: "row", justifyContent: "space-between", alignItems: "center", padding: 15, backgroundColor: "#eee" },
  checkoutBtn: { backgroundColor: "red", padding: 10, borderRadius: 5 },
});
